\section{Lecture One}
\begin{theorem}[Continuity]
  A continuous map on a compact metric space is uniformly continuous.
\end{theorem}
\begin{synexample}[Worked example][worked]
  On $[0,1]$, $\lvert x^2-y^2\rvert\leq 2\lvert x-y\rvert$.
\end{synexample}
\begin{synwarning}[Common pitfall]
  Pointwise choices of $\delta$ do not establish uniform continuity.
\end{synwarning}
\begin{synexercise}[Try it]
  Extract a finite subcover from the continuity neighbourhoods.
\end{synexercise}
\begin{synlecturesummary}
  Continuity transports local information; compactness turns local data into
  finite data. A Lebesgue number converts the finite cover into one global scale.
\end{synlecturesummary}
