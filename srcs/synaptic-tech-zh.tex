\section{架构}

\subsection{模块依赖图}

\begin{verbatim}

  +-- synaptic-base.sty      引擎检查、选项解析

\end{verbatim}

\section{核心命令参考}

\subsection{用户 API}

\begin{description}
  \item[\texttt{\textbackslash SynapticSetup\{key=val\}}] 运行时修改选项。
  \item[\texttt{\textbackslash SynapticTitle\{文本\}}] 文档标题。
  \item[\texttt{\textbackslash SynapticAuthor[标记]\{姓名\}}] 作者。
  \item[\texttt{\textbackslash SynapticAffiliation[标记]\{文本\}}] 附属机构。
  \item[\texttt{\textbackslash SynapticEmail\{地址\}}] 电子邮箱。
  \item[\texttt{\textbackslash SynapticKeywords\{...\}}] 关键词。
  \item[\texttt{\textbackslash SynapticArxiv\{编号\}}] arXiv 链接。
  \item[\texttt{\textbackslash SynapticHeader\{文本\}}] 自定义投稿抬头。
  \item[\texttt{\textbackslash SynapticSubHeader\{文本\}}] 子抬头。
  \item[\texttt{\textbackslash SynapticPreprint\{编号\}}] 预印本标识。
  \item[\texttt{\textbackslash SynapticDedicated\{文本\}}] 致辞文。
  \item[\texttt{\textbackslash SynapticMakeTitle}] 渲染标题页。
  \item[\texttt{\textbackslash SynapticNewTheorem\{环境\}\{名称\}}] 声明定理环境。
  \item[\texttt{\textbackslash SynapticDefineTheme\{主色\}\{辅色\}}] 自定义主题。
\end{description}

\subsection{环境}

\begin{description}
  \item[abstract] 收集摘要文本用于标题页。
  \item[theorem, lemma, proposition, definition] 定理环境（共享计数器，按节编号）。
  \item[proof] 证明环境，含 QED 符号。
  \item[example, remark, warning, exercise] tcolorbox 环境（lecture 模式）。
  \item[synbox] 通用 tcolorbox 带可选标题。
\end{description}

\section{扩展点}

\begin{itemize}
  \item \textbf{新语言} —— 复制 \texttt{synaptic-lang-en.def} 为 \texttt{synaptic-lang-fr.def} 并翻译各标签常量。
  \item \textbf{新主题} —— 使用 \texttt{\textbackslash SynapticDefineTheme\{名\}\{主色\}\{辅色\}}。
  \item \textbf{新字体组合} —— 在 \texttt{synaptic-fonts.sty} 的分发块中新增分支。
  \item \textbf{新模式} —— 创建新的 \texttt{synaptic-*.sty} 并在 \texttt{synaptic-base.sty} 的 mode 分发中添加对应分支。
  \item \textbf{字体自动检测} —— 编辑 \texttt{synaptic-fonts.lua} 修改检测优先级列表。
\end{itemize}

\section{版本历史}

\begin{description}
  \item[v2.0] (2026-06-05) 完全模块化重构：mode 系统、theme 系统、fontset 抽象、Lua 字体检测、非侵入 API、tcolorbox 环境。
  \item[v1.0] (2025) 初始单体版本：单个 \texttt{synaptic.sty} 含基础颜色/样式选项。
\end{description}
