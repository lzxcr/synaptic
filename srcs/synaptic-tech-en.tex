\section{Key Commands Reference}

\subsection{User-Facing API}

\begin{description}
  \item[\texttt{\textbackslash SynapticSetup\{key=val\}}] Runtime option changes.
  \item[\texttt{\textbackslash SynapticTitle\{text\}}] Document title.
  \item[\texttt{\textbackslash SynapticAuthor[mark]\{name\}}] Author with optional affiliation mark.
  \item[\texttt{\textbackslash SynapticAffiliation[mark]\{text\}}] Affiliation.
  \item[\texttt{\textbackslash SynapticEmail\{addr\}}] Email address.
  \item[\texttt{\textbackslash SynapticKeywords\{...\}}] Keywords below abstract.
  \item[\texttt{\textbackslash SynapticArxiv\{ID\}}] arXiv identifier link.
  \item[\texttt{\textbackslash SynapticHeader\{text\}}] Custom submission header.
  \item[\texttt{\textbackslash SynapticSubHeader\{text\}}] Sub-header.
  \item[\texttt{\textbackslash SynapticPreprint\{ID\}}] Preprint identifier.
  \item[\texttt{\textbackslash SynapticDedicated\{text\}}] Dedication text.
  \item[\texttt{\textbackslash SynapticMakeTitle}] Render title page.
  \item[\texttt{\textbackslash SynapticNewTheorem\{env\}\{Name\}}] Declare theorem-like environment.
  \item[\texttt{\textbackslash SynapticDefineTheme\{primary\}\{secondary\}}] Custom colour theme.
\end{description}

\subsection{Environments}

\begin{description}
  \item[abstract] Collect abstract text for title page.
  \item[theorem, lemma, proposition, definition] Theorem environments (shared counter, numbered by section).
  \item[proof] Proof environment with QED symbol.
  \item[example, remark, warning, exercise] tcolorbox environments (lecture mode).
  \item[synbox] Generic tcolorbox with optional title.
\end{description}

\section{Extension Points}

\begin{itemize}
  \item \textbf{New languages} --- Copy \texttt{synaptic-lang-en.def} to \texttt{synaptic-lang-fr.def} and translate each label constant.
  \item \textbf{New themes} --- Use \texttt{\textbackslash SynapticDefineTheme\{name\}\{primary\}\{secondary\}}.
  \item \textbf{New font sets} --- Add a new case to \texttt{synaptic-fonts.sty} in the dispatch block.
  \item \textbf{New modes} --- Create a new \texttt{synaptic-*.sty} file and add a case to the mode dispatch in \texttt{synaptic-base.sty}.
  \item \textbf{Font auto-detect} --- Edit \texttt{synaptic-fonts.lua} to modify the detection priority list.
\end{itemize}
